// ── DEMAND SOURCE: THE P&P TAB ───────────────────────────────────
// Demand is read live from the current "P&P MM.DD" tab (SAP B1 SO export),
// resolved at runtime by resolvePPSheet(). Nothing is hard-coded.
// P&P columns: DPN col F, Open Qty col G, Delivery/Due Date col I; row 1 = headers.
const PP_FIRST_ROW = 2;   // row 1 = headers
const PP_DPN_COL   = 6;   // F  (Decavo Part Number — matches Part Labor Data)
const PP_QTY_COL   = 7;   // G  (Open Qty = demand)
const PP_DUE_COL   = 9;   // I  (Delivery / Due Date)
const PP_DESC_COL  = 0;   // optional item-description column (0 = none)

// ── CONSTANTS ────────────────────────────────────────────────────
const APPLIED_HRS_PER_DAY = 6.75;
const WORK_DAYS_PER_WEEK  = 5;
const AVAIL_HRS_PER_PERSON_PER_WEEK = APPLIED_HRS_PER_DAY * WORK_DAYS_PER_WEEK; // 33.75

const DEPTS = ['LAM','DEPART','CNC TRIM','HAND-TRIM','PLOTTER','QA','KITTING','ASY','PAINT','SHIP'];
const LAM_DEPTS    = ['LAM','DEPART','CNC TRIM','HAND-TRIM','PLOTTER'];
const FINISH_DEPTS = ['QA','KITTING','ASY','PAINT','SHIP'];
const SCHED_DEPTS  = [];  // No per-job ops in the new layout: PLANNING removed, KITTING is now per-part (finish phase)

// Dept colors (hex, no #)
const DEPT_COLORS = {
  'LAM':       '#1F6B75', 'DEPART':    '#2E86AB', 'CNC TRIM':  '#2E7D32',
  'HAND-TRIM': '#558B2F', 'PLOTTER':   '#6D4C41', 'QA':        '#E65100',
  'ASY':       '#C62828', 'PAINT':     '#37474F', 'SHIP':      '#4E342E',
  'PLANNING':  '#6A1B9A', 'KITTING':   '#8E24AA'
};

const COLOR = {
  DARK_BLUE:  '#1F3864', GRAY_HDR:   '#404040', GRAY_SUB:   '#595959',
  CYAN:       '#00CCFF', YELLOW:     '#FFFF00', BLUE_IN:    '#4472C4',
  RED:        '#FF4444', ORANGE:     '#FFA500', GREEN:      '#00B050',
  WHITE:      '#FFFFFF', LIGHT_GRAY: '#F2F2F2', AMBER:      '#FFF8E1',
  RED_LIGHT:  '#FFEBEE', CYAN_LIGHT: '#E0F7FA'

// ── ENTRY POINTS ─────────────────────────────────────────────────

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('📊 Capacity Planning')
    .addItem('🔄 Refresh Capacity', 'refreshCapacity')
    .addItem('🏗️ Build All Sheets (first run)', 'buildAll')
    .addToUi();
}

function removeExistingFilter(ws) {
  try { const f = ws.getFilter(); if (f) f.remove(); } catch(e) {}
}

function buildAll() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  // Note: the "📋 Part Labor Data" tab is user-owned (paste your labor data),
  // and demand is read live from the current "P&P MM.DD" tab. buildAll only
  // (re)generates the derived output sheets.
  buildChartsSheet(ss);
  refreshCapacity();  // reads P&P demand → Demand Detail + Weekly Capacity + LAM Schedule
  SpreadsheetApp.getUi().alert('✅ All sheets built! Click "🔄 Refresh Capacity" any time.');
}

function refreshCapacity() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const laborMap = readLaborData(ss);
  const demand   = readDemandFromPP(ss);   // live demand from the current P&P tab
  const { weeklyHours, lamSchedule, allWeeks } = computeCapacity(laborMap, demand);
  writeWeeklyCapacity(ss, weeklyHours, allWeeks);
  writeLamSchedule(ss, lamSchedule);
  buildDemandSheet(ss, demand);       // rebuild Demand Detail from the live P&P tab
  updateDemandStatus(ss, laborMap);   // then mark labor-data availability
  SpreadsheetApp.getActiveSpreadsheet().toast('✅ Capacity refreshed!', 'Done', 3);
}

// ── READ LABOR DATA FROM SHEET ────────────────────────────────────

function readLaborData(ss) {
  const ws = ss.getSheetByName('📋 Part Labor Data');
  if (!ws) {
    ss.toast('No "📋 Part Labor Data" tab found — paste your labor data into a tab with that name, then Refresh.', 'Missing tab', 6);
    return {};
  }
  const data = ws.getDataRange().getValues();
  // Find header row (row containing "Part Number" in any column) and note which
  // column Part Number lives in — in the new layout it is column B, not A.
  let hdrRow = -1, pnCol = -1;
  for (let i = 0; i < data.length; i++) {
    const idx = data[i].findIndex(c => String(c).trim() === 'Part Number');
    if (idx >= 0) { hdrRow = i; pnCol = idx; break; }
  }
  if (hdrRow < 0) return {};
  const hdrs = data[hdrRow].map(h => String(h).trim());
  const isYes = v => v === true || ['TRUE','YES','X'].indexOf(String(v).trim().toUpperCase()) >= 0;
  const map = {};
  for (let i = hdrRow + 1; i < data.length; i++) {
    const row = data[i];
    const pn = String(row[pnCol]).trim();
    if (!pn) continue;
    const obj = {};
    hdrs.forEach((h, j) => { obj[h] = row[j]; });
    // Only count parts flagged on the Pick & Pack plan (On P&P? = TRUE)
    if (!isYes(obj['On P&P?'])) continue;
    const n = k => { const v = parseFloat(obj[k]); return isNaN(v) ? 0 : v; };
    map[pn] = {
      customer:      String(obj['CUSTOMER'] || ''),
      cyclesPerDay:  n('Tool Cycles / Day') || 1,
      upCount:       n('Parts / Cycle') || 1,
      plotter:       n('PLOTTER'),
      lam:           n('LAM'),
      depart:        n('DEPART'),
      cncTrim:       n('CNC_TRIM'),
      handTrim:      n('HAND_TRIM'),
      qa1:           n('QA_1'),
      qa2:           n('QA_2'),
      qa3:           n('QA_3'),
      qa4:           n('QA_4'),
      kitting:       n('KITTING'),
      asy1:          n('ASY_1'),
      asy2:          n('ASY_2'),
      asyRepair:     n('ASY_REPAIR'),
      outsourceDays: n('OUTSOURCE'),   // business days per MO — shifts LAM start earlier
      paint1:        n('PAINT_1'),
      paint2:        n('PAINT_2'),
      ship:          n('SHIP'),
    };
  }
  return map;
}

// ── READ DEMAND FROM THE P&P TAB ──────────────────────────────────

// Finds the current P&P tab. Its name is "P&P MM.DD" (month.day of the last
// update, no leading zeros); if more than one exists, keep the most recent by
// date. Returns the Sheet or null. (Mirrors the 13-WK Back End resolver so both
// tools always read the same tab.)
function resolvePPSheet(ss) {
  const re = /^P&P\s+(\d{1,2})\.(\d{1,2})$/;
  let best = null, bestKey = -1;
  ss.getSheets().forEach(sh => {
    const m = sh.getName().trim().match(re);
    if (!m) return;
    const key = Number(m[1]) * 100 + Number(m[2]);   // MM.DD → sortable number
    if (key > bestKey) { bestKey = key; best = sh; }
  });
  return best;
}

// Reads demand live from the current P&P tab → [{pn, desc, dueDate:'YYYY-MM-DD', qty}].
// P&P columns: DPN col F, Open Qty col G, Delivery/Due Date col I; row 1 = headers.
function readDemandFromPP(ss) {
  const sh = resolvePPSheet(ss);
  const out = [];
  if (!sh) {
    ss.toast('No "P&P MM.DD" tab found — demand could not be read.', 'Missing P&P tab', 6);
    return out;
  }
  const last = sh.getLastRow();
  if (last < PP_FIRST_ROW) return out;
  const readCols = Math.max(PP_DUE_COL, PP_DPN_COL, PP_QTY_COL, PP_DESC_COL);
  const vals = sh.getRange(PP_FIRST_ROW, 1, last - PP_FIRST_ROW + 1, readCols).getValues();
  for (let i = 0; i < vals.length; i++) {
    const pn = String(vals[i][PP_DPN_COL - 1] || '').trim();
    if (!pn) continue;
    const qty = Number(vals[i][PP_QTY_COL - 1]) || 0;
    if (qty <= 0) continue;
    const dueRaw = vals[i][PP_DUE_COL - 1];
    const due = (dueRaw instanceof Date) ? dueRaw : new Date(dueRaw);
    if (isNaN(due.getTime())) continue;
    const iso = due.getFullYear() + '-' + ('0' + (due.getMonth() + 1)).slice(-2) + '-' + ('0' + due.getDate()).slice(-2);
    const desc = PP_DESC_COL > 0 ? String(vals[i][PP_DESC_COL - 1] || '').trim() : '';
    out.push({ pn: pn, desc: desc, dueDate: iso, qty: qty });
  }
  return out;
}

// ── CORE COMPUTATION ─────────────────────────────────────────────

function getDeptHrs(lab, dept) {
  switch(dept) {
    case 'LAM':      return lab.lam;
    case 'DEPART':   return lab.depart;
    case 'CNC TRIM': return lab.cncTrim;
    case 'HAND-TRIM':return lab.handTrim;
    case 'PLOTTER':  return lab.plotter;
    case 'QA':       return lab.qa1 + lab.qa2 + lab.qa3 + lab.qa4;
    case 'ASY':      return lab.asy1 + lab.asy2 + lab.asyRepair;
    case 'PAINT':    return lab.paint1 + lab.paint2;
    case 'SHIP':     return lab.ship;
    case 'KITTING':  return lab.kitting;
    default: return 0;
  }
}

function mondayOfWeek(dateStr) {
  const d = new Date(dateStr + 'T12:00:00Z');
  const day = d.getUTCDay(); // 0=Sun
  const diff = (day === 0) ? -6 : 1 - day;
  const mon = new Date(d);
  mon.setUTCDate(d.getUTCDate() + diff);
  mon.setUTCHours(0,0,0,0);
  return mon;
}

function weekKey(dateObj) {
  return dateObj.toISOString().slice(0,10);
}

function addWeeks(dateObj, n) {
  const d = new Date(dateObj);
  d.setUTCDate(d.getUTCDate() + n * 7);
  return d;
}

function computeCapacity(laborMap, demand) {
  // weeklyHours: { weekKey: { dept: hours } }
  const weeklyHours = {};
  const lamSchedule = [];

  demand.forEach(row => {
    const lab = laborMap[row.pn];
    if (!lab) return; // no labor data

    const qty          = row.qty;
    const dueWeek      = mondayOfWeek(row.dueDate);
    // OUTSOURCE lead time (business days per MO) pushes upstream work earlier so
    // there is time to outsource before the finish ops. Convert days → whole weeks.
    const outsourceWeeks = lab.outsourceDays > 0 ? Math.ceil(lab.outsourceDays / WORK_DAYS_PER_WEEK) : 0;
    const lamDeadline  = addWeeks(dueWeek, -2 - outsourceWeeks); // last week LAM must be DONE
    const finishWeek   = addWeeks(dueWeek, -1);    // QA/KITTING/ASY/PAINT/SHIP load here

    // ── Cycle constraint: how many weeks of LAM needed? ──
    const partsPerWeek = lab.cyclesPerDay * lab.upCount * WORK_DAYS_PER_WEEK;
    const lamWeeksNeeded = Math.max(1, Math.ceil(qty / partsPerWeek));
    const lamStart     = addWeeks(lamDeadline, -(lamWeeksNeeded - 1));

    // Parts laminated per week (last week gets the remainder)
    const fullWeekQty  = partsPerWeek;
    const lastWeekQty  = qty - fullWeekQty * (lamWeeksNeeded - 1);
    const isPastDeadline = lamDeadline < new Date(); // flag if already overdue

    // ── Spread LAM-group hours across lam weeks ──
    for (let w = 0; w < lamWeeksNeeded; w++) {
      const wk     = addWeeks(lamStart, w);
      const wkKey  = weekKey(wk);
      const wkQty  = (w === lamWeeksNeeded - 1) ? lastWeekQty : fullWeekQty;

      if (!weeklyHours[wkKey]) weeklyHours[wkKey] = {};
      LAM_DEPTS.forEach(dept => {
        const hrs = getDeptHrs(lab, dept) * wkQty;
        weeklyHours[wkKey][dept] = (weeklyHours[wkKey][dept] || 0) + hrs;
      });
      // PLANNING & KITTING only in first week, once per job
      if (w === 0) {
        SCHED_DEPTS.forEach(dept => {
          weeklyHours[wkKey][dept] = (weeklyHours[wkKey][dept] || 0) + getDeptHrs(lab, dept);
        });
      }
    }

    // ── Finish ops in finish week (not spread) ──
    const fKey = weekKey(finishWeek);
    if (!weeklyHours[fKey]) weeklyHours[fKey] = {};
    FINISH_DEPTS.forEach(dept => {
      weeklyHours[fKey][dept] = (weeklyHours[fKey][dept] || 0) + getDeptHrs(lab, dept) * qty;
    });

    // ── LAM schedule row ──
    lamSchedule.push({
      pn:            row.pn,
      desc:          row.desc,
      dueDate:       row.dueDate,
      qty:           qty,
      cyclesPerDay:  lab.cyclesPerDay,
      upCount:       lab.upCount,
      outsourceDays: lab.outsourceDays,
      partsPerWeek:  partsPerWeek,
      lamWeeksNeeded:lamWeeksNeeded,
      lamStart:      weekKey(lamStart),
      lamDeadline:   weekKey(lamDeadline),
      finishWeek:    weekKey(finishWeek),
      hasLaborData:  true,
      overloaded:    isPastDeadline || lamStart < new Date('2026-01-01') // flag if start is in past
    });
  });

  // Flag demand rows with no labor data
  demand.forEach(row => {
    if (!laborMap[row.pn]) {
      const dueWeek = mondayOfWeek(row.dueDate);
      lamSchedule.push({
        pn: row.pn, desc: row.desc, dueDate: row.dueDate, qty: row.qty,
        cyclesPerDay: '?', upCount: '?', outsourceDays: '', partsPerWeek: '?',
        lamWeeksNeeded: '?', lamStart: '?', lamDeadline: weekKey(addWeeks(dueWeek,-2)),
        finishWeek: weekKey(addWeeks(dueWeek,-1)), hasLaborData: false, overloaded: false
      });
    }
  });

  // Sorted unique weeks
  const allWeeks = Object.keys(weeklyHours).sort();
  return { weeklyHours, lamSchedule, allWeeks };
}

// ── WRITE WEEKLY CAPACITY SHEET ───────────────────────────────────

function writeWeeklyCapacity(ss, weeklyHours, allWeeks) {
  let ws = ss.getSheetByName('📊 Weekly Capacity');
  if (!ws) ws = ss.insertSheet('📊 Weekly Capacity');

  // Preserve headcount inputs (row 5, cols 2+) before wiping the sheet
  let savedHeadcounts = null;
  try {
    const existing = ws.getRange(5, 2, 1, DEPTS.length).getValues()[0];
    if (existing.some(v => typeof v === 'number' && v > 0)) {
      savedHeadcounts = existing;
    }
  } catch(e) {}

  removeExistingFilter(ws);
  ws.clearContents();
  ws.clearFormats();

  const numDepts = DEPTS.length;
  const totalCols = numDepts + 1;

  // Row 1 — Title
  ws.getRange(1, 1, 1, totalCols).merge()
    .setValue('WEEKLY LABOR DEMAND BY DEPARTMENT — Cycle-Constrained Capacity Plan')
    .setBackground(COLOR.DARK_BLUE).setFontColor(COLOR.WHITE)
    .setFontWeight('bold').setFontSize(13).setHorizontalAlignment('center');
  ws.setRowHeight(1, 32);

  // Row 2 — Note
  ws.getRange(2, 1, 1, totalCols).merge()
    .setValue('LAM/TRIM/PLOTTER hours spread backward from due date based on cycle rate & parts/cycle  |  QA/KITTING/ASY/PAINT/SHIP load 1 wk before due  |  OUTSOURCE days shift LAM start earlier  |  Edit Part Labor Data → click Refresh')
    .setBackground('#E8F5E9').setFontColor('#333333').setFontStyle('italic').setFontSize(9)
    .setHorizontalAlignment('center');
  ws.setRowHeight(2, 16);

  // Row 3 — headcount inputs label
  ws.getRange(3, 1, 1, totalCols).merge()
    .setValue('HEADCOUNT INPUTS — edit blue cells below (avail hrs/person/week = ' + AVAIL_HRS_PER_PERSON_PER_WEEK + ' hrs)')
    .setBackground(COLOR.GRAY_HDR).setFontColor(COLOR.WHITE)
    .setFontWeight('bold').setFontSize(10).setHorizontalAlignment('center');
  ws.setRowHeight(3, 20);

  // Row 4 — dept headers
  ws.getRange(4, 1).setValue('Week').setBackground(COLOR.GRAY_HDR).setFontColor(COLOR.WHITE).setFontWeight('bold').setHorizontalAlignment('center');
  DEPTS.forEach((dept, i) => {
    ws.getRange(4, i + 2).setValue(dept)
      .setBackground(DEPT_COLORS[dept]).setFontColor(COLOR.WHITE)
      .setFontWeight('bold').setFontSize(9).setHorizontalAlignment('center');
  });
  ws.setRowHeight(4, 24);

  // Row 5 — headcount inputs (blue) — restore saved values if present, else default to 1
  ws.getRange(5, 1).setValue('# Employees').setBackground(COLOR.BLUE_IN).setFontColor(COLOR.WHITE).setFontWeight('bold').setHorizontalAlignment('center');
  DEPTS.forEach((dept, i) => {
    const hc = (savedHeadcounts && savedHeadcounts[i]) ? savedHeadcounts[i] : 1;
    ws.getRange(5, i + 2).setValue(hc)
      .setBackground(COLOR.BLUE_IN).setFontColor(COLOR.WHITE)
      .setFontWeight('bold').setHorizontalAlignment('center');
  });
  ws.setRowHeight(5, 22);

  // Row 6 — capacity formula
  ws.getRange(6, 1).setValue('Avail Hrs/Wk').setBackground('#E3F2FD').setFontColor('#1565C0').setFontStyle('italic').setHorizontalAlignment('center');
  const colLetters = ['B','C','D','E','F','G','H','I','J','K','L','M','N','O','P','Q','R','S','T'];
  DEPTS.forEach((dept, i) => {
    const cl = colLetters[i];
    ws.getRange(6, i + 2).setFormula('=' + cl + '5*' + AVAIL_HRS_PER_PERSON_PER_WEEK)
      .setBackground('#E3F2FD').setFontColor('#1565C0')
      .setNumberFormat('0.0').setHorizontalAlignment('center');
  });
  ws.setRowHeight(6, 18);

  // Row 7 — spacer
  ws.setRowHeight(7, 6);

  // Row 8 — column sub-headers
  ws.getRange(8, 1).setValue('Week of').setBackground(COLOR.GRAY_SUB||'#595959').setFontColor(COLOR.WHITE).setFontWeight('bold').setHorizontalAlignment('center');
  DEPTS.forEach((dept, i) => {
    ws.getRange(8, i + 2).setValue('Demand Hrs')
      .setBackground(COLOR.GRAY_SUB||'#595959').setFontColor(COLOR.WHITE)
      .setFontWeight('bold').setFontSize(9).setHorizontalAlignment('center');
  });
  ws.setRowHeight(8, 18);

  // Data rows starting row 9
  const dataStartRow = 9;
  allWeeks.forEach((wk, rowIdx) => {
    const r = dataStartRow + rowIdx;
    const isAlt = rowIdx % 2 === 0;
    const rowBg = isAlt ? '#F0F4FF' : COLOR.WHITE;

    ws.getRange(r, 1).setValue(wk).setBackground('#E8EAF6').setFontWeight('bold').setHorizontalAlignment('center');

    DEPTS.forEach((dept, i) => {
      const hrs = (weeklyHours[wk] && weeklyHours[wk][dept]) ? weeklyHours[wk][dept] : 0;
      const cell = ws.getRange(r, i + 2);
      cell.setValue(Math.round(hrs * 10) / 10).setNumberFormat('0.0').setHorizontalAlignment('center');

      // Conditional: red if demand > capacity (compare to row 6)
      if (hrs === 0) {
        cell.setBackground(rowBg).setFontColor('#CCCCCC');
      } else {
        cell.setBackground(rowBg).setFontColor('#000000');
      }
    });
    ws.setRowHeight(r, 15);
  });

  // Totals row
  const totalRow = dataStartRow + allWeeks.length;
  ws.getRange(totalRow, 1).setValue('TOTAL').setBackground(COLOR.GRAY_HDR).setFontColor(COLOR.WHITE).setFontWeight('bold').setHorizontalAlignment('center');
  DEPTS.forEach((dept, i) => {
    const cl = colLetters[i];
    ws.getRange(totalRow, i + 2)
      .setFormula('=SUM(' + cl + dataStartRow + ':' + cl + (totalRow-1) + ')')
      .setBackground(COLOR.GRAY_HDR).setFontColor(COLOR.WHITE).setFontWeight('bold')
      .setNumberFormat('0.0').setHorizontalAlignment('center');
  });
  ws.setRowHeight(totalRow, 18);

  // Column widths
  ws.setColumnWidth(1, 110);
  for (let i = 2; i <= totalCols; i++) ws.setColumnWidth(i, 90);

  // Freeze
  ws.setFrozenRows(8);

  // Apply conditional formatting: red background if demand > capacity
  // We do this with a note — Google Sheets conditional formatting via script
  const dataRange = ws.getRange(dataStartRow, 2, allWeeks.length, numDepts);
  const rules = [];
  DEPTS.forEach((dept, i) => {
    const col = colLetters[i];
    const capCell = '$' + col + '$6';
    const range = ws.getRange(dataStartRow, i + 2, allWeeks.length, 1);
    const rule = SpreadsheetApp.newConditionalFormatRule()
      .whenFormulaSatisfied('=' + col + dataStartRow + '>' + capCell)
      .setBackground('#FFCDD2')
      .setFontColor('#B71C1C')
      .setRanges([range])
      .build();
    rules.push(rule);
  });
  ws.setConditionalFormatRules(rules);
}

// ── WRITE LAM SCHEDULE SHEET ──────────────────────────────────────

function writeLamSchedule(ss, lamSchedule) {
  let ws = ss.getSheetByName('🗓️ LAM Schedule');
  if (!ws) ws = ss.insertSheet('🗓️ LAM Schedule');
  removeExistingFilter(ws);
  ws.clearContents();
  ws.clearFormats();

  const headers = ['Part Number','Description','Due Date','Open QTY','Outsrc Days',
                   'Cycles/Day','Parts/Cycle','Parts/Wk','LAM Wks Needed',
                   'LAM Start','LAM Deadline','Finish Ops Wk','Status'];
  const colWidths = [110,280,100,80,80,80,80,80,110,100,110,110,180];

  // Title
  ws.getRange(1, 1, 1, headers.length).merge()
    .setValue('LAM SCHEDULE — Cycle-Constrained Start Dates')
    .setBackground(COLOR.DARK_BLUE).setFontColor(COLOR.WHITE)
    .setFontWeight('bold').setFontSize(13).setHorizontalAlignment('center');
  ws.setRowHeight(1, 30);

  ws.getRange(2, 1, 1, headers.length).merge()
    .setValue('🔴 RED = LAM start date is in the past — already behind schedule  |  🔵 CYAN = missing labor data  |  Sorted by Due Date')
    .setBackground('#FFF3CD').setFontColor('#555555').setFontStyle('italic').setFontSize(9)
    .setHorizontalAlignment('left');
  ws.setRowHeight(2, 16);

  // Header row
  headers.forEach((h, i) => {
    ws.getRange(3, i + 1).setValue(h)
      .setBackground(COLOR.GRAY_HDR).setFontColor(COLOR.WHITE)
      .setFontWeight('bold').setFontSize(9).setHorizontalAlignment('center');
    ws.setColumnWidth(i + 1, colWidths[i]);
  });
  ws.setRowHeight(3, 32);

  // Sort by due date
  const sorted = lamSchedule.slice().sort((a, b) => a.dueDate.localeCompare(b.dueDate));
  const today = new Date().toISOString().slice(0,10);

  sorted.forEach((row, idx) => {
    const r = idx + 4;
    const isAlt = idx % 2 === 0;
    const rowBg = isAlt ? '#F5F5F5' : COLOR.WHITE;

    let status, statusBg, statusFg, rowHighlight;
    if (!row.hasLaborData) {
      status = '⚠ NO LABOR DATA'; statusBg = COLOR.CYAN; statusFg = '#003355'; rowHighlight = COLOR.CYAN;
    } else if (row.lamStart < today && row.lamStart !== '?') {
      status = '🔴 LAM START IN PAST'; statusBg = COLOR.RED; statusFg = COLOR.WHITE; rowHighlight = '#FFEBEE';
    } else {
      status = '✅ On track'; statusBg = '#00B050'; statusFg = COLOR.WHITE; rowHighlight = rowBg;
    }

    const vals = [
      row.pn, row.desc, row.dueDate, row.qty, row.outsourceDays,
      row.cyclesPerDay, row.upCount, row.partsPerWeek, row.lamWeeksNeeded,
      row.lamStart, row.lamDeadline, row.finishWeek, status
    ];

    vals.forEach((v, i) => {
      const cell = ws.getRange(r, i + 1);
      cell.setValue(v).setBackground(rowHighlight).setFontSize(9);
      if (i === 0) cell.setFontWeight('bold');
      if (i === 12) { cell.setBackground(statusBg).setFontColor(statusFg).setFontWeight('bold'); }
      cell.setHorizontalAlignment(i <= 1 ? 'left' : 'center');
    });

  });

  ws.setFrozenRows(3);

  // Auto-filter
  removeExistingFilter(ws);
  ws.getRange(3, 1, 1, headers.length).createFilter();
}

// ── UPDATE DEMAND STATUS ──────────────────────────────────────────

function updateDemandStatus(ss, laborMap) {
  const ws = ss.getSheetByName('📦 Demand Detail');
  if (!ws) return;
  const data = ws.getDataRange().getValues();
  // Find header row
  let hdrRow = -1;
  for (let i = 0; i < data.length; i++) {
    if (String(data[i][0]).trim() === 'Part Number') { hdrRow = i; break; }
  }
  if (hdrRow < 0) return;
  const pnCol = 0, statusCol = data[hdrRow].findIndex(h => String(h).trim() === 'Status');
  if (statusCol < 0) return;
  for (let i = hdrRow + 1; i < data.length; i++) {
    const pn = String(data[i][pnCol]).trim();
    if (!pn) continue;
    const hasLab = !!laborMap[pn];
    ws.getRange(i + 1, statusCol + 1)
      .setValue(hasLab ? '✅ Has Labor Data' : '⚠ MISSING LABOR DATA')
      .setBackground(hasLab ? '#00B050' : '#00CCFF')
      .setFontColor(hasLab ? COLOR.WHITE : '#003355')
      .setFontWeight('bold');
  }
}

// ── CUSTOM FUNCTIONS: ON P&P? ─────────────────────────────────────
// Populates the "On P&P?" column. Put this in each row of column A of the
// Part Labor Data sheet, e.g. in A9 (first data row):
//   =IS_ON_PP_OR_ANCESTOR(B9, PP_LIST(TODAY()), 'Parent to Child'!A2:C)
// then fill down. readLaborData() only counts parts whose On P&P? = TRUE.
// PP_LIST(TODAY()) auto-locates the current P&P tab, so the daily rename of
// that tab never breaks these formulas.

/**
 * Column F (DPNs, row 2 down) of the CURRENT "P&P MM.DD" tab — resolved by
 * resolvePPSheet() (most recent dated tab), so the daily rename never breaks
 * the formula. Pass TODAY() as refreshToken so the value recomputes daily
 * (custom functions cache until an argument changes).
 * @param {number} refreshToken  Pass TODAY() to force a daily refresh.
 * @return {Array} Column F values as a 2-D array, for use as ppRange.
 * @customfunction
 */
function PP_LIST(refreshToken) {
  const pp = resolvePPSheet(SpreadsheetApp.getActiveSpreadsheet());
  if (!pp) throw new Error('No "P&P MM.DD" tab was found.');
  const last = pp.getLastRow();
  if (last < PP_FIRST_ROW) return [['']];
  return pp.getRange(PP_FIRST_ROW, PP_DPN_COL, last - PP_FIRST_ROW + 1, 1).getValues();
}

/**
 * TRUE if a DPN — or any ancestor (parent, grandparent, …) — is on the
 * Pick & Pack list.
 * @param {string} dpn               Part number to check (e.g. D50567)
 * @param {Array}  ppRange           P&P list column, e.g. PP_LIST(TODAY())
 * @param {Array}  parentChildRange  Parent→Child table, e.g. 'Parent to Child'!A2:C (A=parent, B=child)
 * @return {boolean}
 * @customfunction
 */
function IS_ON_PP_OR_ANCESTOR(dpn, ppRange, parentChildRange) {
  var key = function(v) { return (v === null || v === undefined) ? '' : String(v).trim(); };
  var start = key(dpn);
  if (!start) return false;

  // A single-cell argument arrives as a scalar, not a 2-D array — normalize it.
  if (!Array.isArray(ppRange)) ppRange = [[ppRange]];
  if (!Array.isArray(parentChildRange)) parentChildRange = [[parentChildRange]];

  var ppSet = {};
  ppRange.forEach(function(row) { var k = key(row[0]); if (k) ppSet[k] = true; });

  var childToParents = {};
  parentChildRange.forEach(function(row) {
    var parent = key(row[0]), child = key(row[1]);
    if (!parent || !child) return;
    (childToParents[child] = childToParents[child] || []).push(parent);
  });

  var visited = {};
  function checkAncestors(pn) {
    if (ppSet[pn]) return true;
    if (visited[pn]) return false; // guards against circular BOM data
    visited[pn] = true;
    var parents = childToParents[pn] || [];
    for (var i = 0; i < parents.length; i++) {
      if (checkAncestors(parents[i])) return true;
    }
    return false;
  }
  return checkAncestors(start);
}
