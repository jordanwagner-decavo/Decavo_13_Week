function readLaborData(ss) {
  const ws = ss.getSheetByName('📋 Part Labor Data');
  if (!ws) { reseedLaborData(ss); return readLaborData(ss); }
  const data = ws.getDataRange().getValues();
  // Find header row (row containing "Part Number" in any column) and note which
  // column Part Number lives in — in the new layout it is column B, not A.
  let hdrRow = -1, pnCol = -1;
  for (let i = 0; i < data.length; i++) {
    const idx = data[i].findIndex(c => String(c).trim() === 'Part Number');
    if (idx >= 0) { hdrRow = i; pnCol = idx; break; }
  }
  if (hdrRow < 0) return {};
  const hdrs = data[hdrRow].map(h => String(h).trim());
  const isYes = v => v === true || ['TRUE','YES','X'].indexOf(String(v).trim().toUpperCase()) >= 0;
  const map = {};
  for (let i = hdrRow + 1; i < data.length; i++) {
    const row = data[i];
    const pn = String(row[pnCol]).trim();
    if (!pn) continue;
    const obj = {};
    hdrs.forEach((h, j) => { obj[h] = row[j]; });
    // Only count parts flagged on the Pick & Pack plan (On P&P? = TRUE)
    if (!isYes(obj['On P&P?'])) continue;
    const n = k => { const v = parseFloat(obj[k]); return isNaN(v) ? 0 : v; };
    map[pn] = {
      customer:      String(obj['CUSTOMER'] || ''),
      cyclesPerDay:  n('Tool Cycles / Day') || 1,
      upCount:       n('Parts / Cycle') || 1,
      plotter:       n('PLOTTER'),
      lam:           n('LAM'),
      depart:        n('DEPART'),
      cncTrim:       n('CNC_TRIM'),
      handTrim:      n('HAND_TRIM'),
      qa1:           n('QA_1'),
      qa2:           n('QA_2'),
      qa3:           n('QA_3'),
      qa4:           n('QA_4'),
      kitting:       n('KITTING'),
      asy1:          n('ASY_1'),
      asy2:          n('ASY_2'),
      asyRepair:     n('ASY_REPAIR'),
      outsourceDays: n('OUTSOURCE'),   // business days per MO — shifts LAM start earlier
      paint1:        n('PAINT_1'),
      paint2:        n('PAINT_2'),
      ship:          n('SHIP'),
    };
  }
  return map;
}

// ── CORE COMPUTATION ─────────────────────────────────────────────

function getDeptHrs(lab, dept) {
  switch(dept) {
    case 'LAM':      return lab.lam;
    case 'DEPART':   return lab.depart;
    case 'CNC TRIM': return lab.cncTrim;
    case 'HAND-TRIM':return lab.handTrim;
    case 'PLOTTER':  return lab.plotter;
    case 'QA':       return lab.qa1 + lab.qa2 + lab.qa3 + lab.qa4;
    case 'ASY':      return lab.asy1 + lab.asy2 + lab.asyRepair;
    case 'PAINT':    return lab.paint1 + lab.paint2;
    case 'SHIP':     return lab.ship;
    case 'KITTING':  return lab.kitting;
    default: return 0;
  }
}

function mondayOfWeek(dateStr) {
  const d = new Date(dateStr + 'T12:00:00Z');
  const day = d.getUTCDay(); // 0=Sun
  const diff = (day === 0) ? -6 : 1 - day;
  const mon = new Date(d);
  mon.setUTCDate(d.getUTCDate() + diff);
  mon.setUTCHours(0,0,0,0);
  return mon;
}

function weekKey(dateObj) {
  return dateObj.toISOString().slice(0,10);
}

function addWeeks(dateObj, n) {
  const d = new Date(dateObj);
  d.setUTCDate(d.getUTCDate() + n * 7);
  return d;
}

function computeCapacity(laborMap) {
  // weeklyHours: { weekKey: { dept: hours } }
  const weeklyHours = {};
  const lamSchedule = [];

  DEMAND_DATA.forEach(row => {
    const lab = laborMap[row.pn];
    if (!lab) return; // no labor data

    const qty          = row.qty;
    const dueWeek      = mondayOfWeek(row.dueDate);
    // OUTSOURCE lead time (business days per MO) pushes upstream work earlier so
    // there is time to outsource before the finish ops. Convert days → whole weeks.
    const outsourceWeeks = lab.outsourceDays > 0 ? Math.ceil(lab.outsourceDays / WORK_DAYS_PER_WEEK) : 0;
    const lamDeadline  = addWeeks(dueWeek, -2 - outsourceWeeks); // last week LAM must be DONE
    const finishWeek   = addWeeks(dueWeek, -1);    // QA/KITTING/ASY/PAINT/SHIP load here

    // ── Cycle constraint: how many weeks of LAM needed? ──
    const partsPerWeek = lab.cyclesPerDay * lab.upCount * WORK_DAYS_PER_WEEK;
    const lamWeeksNeeded = Math.max(1, Math.ceil(qty / partsPerWeek));
    const lamStart     = addWeeks(lamDeadline, -(lamWeeksNeeded - 1));

    // Parts laminated per week (last week gets the remainder)
    const fullWeekQty  = partsPerWeek;
    const lastWeekQty  = qty - fullWeekQty * (lamWeeksNeeded - 1);
    const isPastDeadline = lamDeadline < new Date(); // flag if already overdue

    // ── Spread LAM-group hours across lam weeks ──
    for (let w = 0; w < lamWeeksNeeded; w++) {
      const wk     = addWeeks(lamStart, w);
      const wkKey  = weekKey(wk);
      const wkQty  = (w === lamWeeksNeeded - 1) ? lastWeekQty : fullWeekQty;

      if (!weeklyHours[wkKey]) weeklyHours[wkKey] = {};
      LAM_DEPTS.forEach(dept => {
        const hrs = getDeptHrs(lab, dept) * wkQty;
        weeklyHours[wkKey][dept] = (weeklyHours[wkKey][dept] || 0) + hrs;
      });
      // PLANNING & KITTING only in first week, once per job
      if (w === 0) {
        SCHED_DEPTS.forEach(dept => {
          weeklyHours[wkKey][dept] = (weeklyHours[wkKey][dept] || 0) + getDeptHrs(lab, dept);
        });
      }
    }

    // ── Finish ops in finish week (not spread) ──
    const fKey = weekKey(finishWeek);
    if (!weeklyHours[fKey]) weeklyHours[fKey] = {};
    FINISH_DEPTS.forEach(dept => {
      weeklyHours[fKey][dept] = (weeklyHours[fKey][dept] || 0) + getDeptHrs(lab, dept) * qty;
    });

    // ── LAM schedule row ──
    lamSchedule.push({
      pn:            row.pn,
      desc:          row.desc,
      dueDate:       row.dueDate,
      qty:           qty,
      cyclesPerDay:  lab.cyclesPerDay,
      upCount:       lab.upCount,
      outsourceDays: lab.outsourceDays,
      partsPerWeek:  partsPerWeek,
      lamWeeksNeeded:lamWeeksNeeded,
      lamStart:      weekKey(lamStart),
      lamDeadline:   weekKey(lamDeadline),
      finishWeek:    weekKey(finishWeek),
      hasLaborData:  true,
      overloaded:    isPastDeadline || lamStart < new Date('2026-01-01') // flag if start is in past
    });
  });

  // Flag demand rows with no labor data
  DEMAND_DATA.forEach(row => {
    if (!laborMap[row.pn]) {
      const dueWeek = mondayOfWeek(row.dueDate);
      lamSchedule.push({
        pn: row.pn, desc: row.desc, dueDate: row.dueDate, qty: row.qty,
        cyclesPerDay: '?', upCount: '?', outsourceDays: '', partsPerWeek: '?',
        lamWeeksNeeded: '?', lamStart: '?', lamDeadline: weekKey(addWeeks(dueWeek,-2)),
        finishWeek: weekKey(addWeeks(dueWeek,-1)), hasLaborData: false, overloaded: false
      });
    }
  });

  // Sorted unique weeks
  const allWeeks = Object.keys(weeklyHours).sort();
  return { weeklyHours, lamSchedule, allWeeks };
}

// ── UPDATE DEMAND STATUS ──────────────────────────────────────────

function updateDemandStatus(ss, laborMap) {
  const ws = ss.getSheetByName('📦 Demand Detail');
  if (!ws) return;
  const data = ws.getDataRange().getValues();
  // Find header row
  let hdrRow = -1;
  for (let i = 0; i < data.length; i++) {
    if (String(data[i][0]).trim() === 'Part Number') { hdrRow = i; break; }
  }
  if (hdrRow < 0) return;
  const pnCol = 0, statusCol = data[hdrRow].findIndex(h => String(h).trim() === 'Status');
  if (statusCol < 0) return;
  for (let i = hdrRow + 1; i < data.length; i++) {
    const pn = String(data[i][pnCol]).trim();
    if (!pn) continue;
    const hasLab = !!laborMap[pn];
    ws.getRange(i + 1, statusCol + 1)
      .setValue(hasLab ? '✅ Has Labor Data' : '⚠ MISSING LABOR DATA')
      .setBackground(hasLab ? '#00B050' : '#00CCFF')
      .setFontColor(hasLab ? COLOR.WHITE : '#003355')
      .setFontWeight('bold');
  }
}

// True if the Part Labor Data tab already has a Part Number header with at
// least one populated part row below it (used to avoid clobbering pasted data).
function laborSheetHasData(ws) {
  const data = ws.getDataRange().getValues();
  for (let i = 0; i < data.length; i++) {
    const idx = data[i].findIndex(c => String(c).trim() === 'Part Number');
    if (idx >= 0) {
      for (let j = i + 1; j < data.length; j++) {
        if (String(data[j][idx]).trim() !== '') return true;
      }
      return false;
    }
  }
  return false;
}

// ── CUSTOM FUNCTIONS: ON P&P? ─────────────────────────────────────
// Populates the "On P&P?" column. Put this in each row of column A of the
// Part Labor Data sheet, e.g. in A9 (first data row):
//   =IS_ON_PP_OR_ANCESTOR(B9, PP_LIST(TODAY()), 'Parent to Child'!A2:C)
// then fill down. readLaborData() only counts parts whose On P&P? = TRUE.
// PP_LIST(TODAY()) auto-locates the current P&P tab, so the daily rename of
// that tab never breaks these formulas.

// Live "On P&P?" formula for a given sheet row (Part Number is in column B).
function ppOnPpFormula(row) {
  return "=IS_ON_PP_OR_ANCESTOR(B" + row + ", PP_LIST(TODAY()), 'Parent to Child'!A2:C)";
}

/**
 * Column F (DPNs, row 2 down) of the CURRENT Pick & Pack tab — the sheet whose
 * name starts with "P&P". The P&P tab is renamed with a new date each day, so
 * this avoids hard-coding the date in a formula. Pass TODAY() as refreshToken
 * so the value recomputes daily (custom functions cache until an arg changes).
 * @param {number} refreshToken  Pass TODAY() to force a daily refresh.
 * @return {Array} Column F values as a 2-D array, for use as ppRange.
 * @customfunction
 */
function PP_LIST(refreshToken) {
  const sheets = SpreadsheetApp.getActiveSpreadsheet().getSheets();
  let pp = null;
  for (let i = 0; i < sheets.length; i++) {
    if (sheets[i].getName().trim().toUpperCase().indexOf('P&P') === 0) { pp = sheets[i]; break; }
  }
  if (!pp) throw new Error('No tab whose name starts with "P&P" was found.');
  const last = pp.getLastRow();
  if (last < 2) return [['']];
  return pp.getRange(2, 6, last - 1, 1).getValues();  // column F = 6
}

/**
 * TRUE if a DPN — or any ancestor (parent, grandparent, …) — is on the
 * Pick & Pack list.
 * @param {string} dpn               Part number to check (e.g. D50567)
 * @param {Array}  ppRange           P&P list column, e.g. PP_LIST(TODAY())
 * @param {Array}  parentChildRange  Parent→Child table, e.g. 'Parent to Child'!A2:C (A=parent, B=child)
 * @return {boolean}
 * @customfunction
 */
function IS_ON_PP_OR_ANCESTOR(dpn, ppRange, parentChildRange) {
  var key = function(v) { return (v === null || v === undefined) ? '' : String(v).trim(); };
  var start = key(dpn);
  if (!start) return false;

  // A single-cell argument arrives as a scalar, not a 2-D array — normalize it.
  if (!Array.isArray(ppRange)) ppRange = [[ppRange]];
  if (!Array.isArray(parentChildRange)) parentChildRange = [[parentChildRange]];

  var ppSet = {};
  ppRange.forEach(function(row) { var k = key(row[0]); if (k) ppSet[k] = true; });

  var childToParents = {};
  parentChildRange.forEach(function(row) {
    var parent = key(row[0]), child = key(row[1]);
    if (!parent || !child) return;
    (childToParents[child] = childToParents[child] || []).push(parent);
  });

  var visited = {};
  function checkAncestors(pn) {
    if (ppSet[pn]) return true;
    if (visited[pn]) return false; // guards against circular BOM data
    visited[pn] = true;
    var parents = childToParents[pn] || [];
    for (var i = 0; i < parents.length; i++) {
      if (checkAncestors(parents[i])) return true;
    }
    return false;
  }
  return checkAncestors(start);
}
