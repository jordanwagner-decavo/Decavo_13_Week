    try {
      DATA = d; CWI = (d.currentWeekIndex == null ? -1 : d.currentWeekIndex);
      DATA.rows.forEach(function (r) {
        r.gospel = r.qtys.reduce(function (a, q) { return a + (q === '' ? 0 : Number(q)); }, 0);

      pending.clear(); savedKeys.clear();
      sortState = { key:null, dir:'asc' };

      console.error('onData failed:', err);

    } finally {
      showOverlay(false);
    }
  }

  function frac(a, b) { return a + '<span class="sl">/</span>' + b; }

  function renderCell(td, ri, wi) {
    var k = ri + ':' + wi;
    td.classList.remove('pending', 'saved', 'empty');
    var q;
    if (pending.has(k)) { td.classList.add('pending'); q = pending.get(k).newQty; }
    else {
      q = DATA.rows[ri].qtys[wi];
      if (q === '') td.classList.add('empty');
      if (savedKeys.has(k)) td.classList.add('saved');
    }
    var price = DATA.rows[ri].unitPrice || 0;
    var pQty  = (q === '' ? 0 : Number(q));
    var qDisp = (q === '' ? '·' : esc(String(q)));
    var act   = isActualWeek(wi) && !pending.has(k);  // pending edits show the new plan, not actuals
    var aQty  = (DATA.rows[ri].actualQtys || [])[wi] || 0;
    var aVal  = (DATA.rows[ri].actualVals || [])[wi] || 0;

    var qtyHtml = act ? frac(String(aQty), (q === '' ? '0' : String(pQty))) : qDisp;
    var pVal = pQty * price;
    var valHtml;
    if (act) valHtml = (aVal === 0 && pVal === 0) ? '' : frac(valueK(aVal), valueK(pVal));
    else     valHtml = (pVal === 0) ? '' : valueK(pVal);
    var past = act && wi < CWI;
    var cqCls = past ? (aQty >= pQty ? ' ok' : ' under') : '';

    // Current/past weeks with nothing planned and nothing shipped read as blank.
    if (act && aQty === 0 && pQty === 0) {
      td.classList.add('empty');
      td.innerHTML = '<div class="cell-pad">·</div>';
      return;
    }

    if (showValues || act) {
      var lines = '<span class="cq' + cqCls + '">' + qtyHtml + '</span>';
      if (showValues && valHtml) lines += '<span class="cv' + cqCls + '">' + valHtml + '</span>';
      td.innerHTML = '<div class="cell-pad cell-dual">' + lines + '</div>';
    } else {
      td.innerHTML = '<div class="cell-pad">' + qDisp + '</div>';
    }
  }

  // Effective planned qty for a cell (pending edit wins).
  function effQty(ri, wi) {
    var k = ri + ':' + wi;
    if (pending.has(k)) return pending.get(k).newQty;
    var q = DATA.rows[ri].qtys[wi];
    return q === '' ? 0 : Number(q);
  }
  // Open miss for a row: sum over locked past weeks of max(plan - actual, 0).
  function rowMiss(ri) {
    if (CWI < 0) return 0;
    var miss = 0;
    for (var wi = 0; wi < CWI; wi++) {
      var plan = DATA.rows[ri].qtys[wi]; plan = (plan === '' ? 0 : Number(plan));
      var act  = (DATA.rows[ri].actualQtys || [])[wi] || 0;
      if (act < plan) miss += (plan - act);
    }
    return miss;
  }
  // Total delivery qty = shipped actuals (past) + planned (current & future).
  // Gospel total comes from the sheet at load and never changes.
  function rowTotal(ri) {
    var g = DATA.rows[ri].gospel || 0;
    return { qty: g, val: g * (DATA.rows[ri].unitPrice || 0) };
  }
  // Sum accounted for: shipped actuals (past) + planned (current + future).
  function effectiveSum(ri) {
    var q = 0;
    DATA.weeks.forEach(function (wk, wi) {
      if (CWI >= 0 && wi < CWI) q += (DATA.rows[ri].actualQtys || [])[wi] || 0;
      else q += effQty(ri, wi);
    });
    return q;
  }
  function renderTotal(td, ri) {
    var t = rowTotal(ri);
    var diff = t.qty - effectiveSum(ri);   // >0 short → push; <0 over → excess
    td.classList.toggle('warn', diff !== 0);
    var note = diff > 0 ? ('⚠ ' + diff + ' to push') : (diff < 0 ? ('⚠ ' + Math.abs(diff) + ' in excess') : '');
    var qd = String(t.qty);
    if (showValues) {
      td.innerHTML = '<div class="cell-pad cell-dual"><span class="cq">' + qd + '</span><span class="cv">' + valueK(t.val) + '</span>' +
        (note ? '<span class="miss">' + note + '</span>' : '') + '</div>';
    } else {
      td.innerHTML = '<div class="cell-pad"><span>' + qd + '</span>' + (note ? '<span class="miss">' + note + '</span>' : '') + '</div>';
    }
  }
  // How much of this row's pending future bumps counts as miss recovery.
  function recoveredForRow(ri) {
    var bump = 0;
    pending.forEach(function (e) {
      if (e.ri === ri && e.wi >= CWI) { var d = e.newQty - e.oldQty; if (d > 0) bump += d; }
    });
    return bump;
  }
  // Per-week shortfalls for a row (locked past weeks), oldest first.
  function missWeeks(ri) {
    var out = [];
    if (CWI < 0) return out;
    for (var wi = 0; wi < CWI; wi++) {
      var plan = DATA.rows[ri].qtys[wi]; plan = (plan === '' ? 0 : Number(plan));
      var act  = (DATA.rows[ri].actualQtys || [])[wi] || 0;
      if (act < plan) out.push({ wi: wi, weekDate: DATA.weeks[wi], qty: plan - act });
    }
    return out;
  }

  // Weeks where cumulative SO demand-due exceeds cumulative supply (behind).
  // Supply = actuals(past) + planned/pending(current+future). Demand from P&P
  // Open Qty, bucketed by due date into the first week ending on/after the due date.
  function behindWeeks(ri) {
    var flags = {};
    var dem = DATA.rows[ri].demand || [];
    if (!dem.length || !DATA.weeks.length) return flags;
    var nW = DATA.weeks.length;
    var weekEnd = DATA.weeks.map(function (d) { return new Date(d).getTime(); });
    // demand due in each week bucket
    var demByWeek = new Array(nW).fill(0);
    dem.forEach(function (d) {
      var t = new Date(d.due).getTime();
      var wi = 0;
      while (wi < nW && weekEnd[wi] < t) wi++;
      if (wi >= nW) wi = nW - 1;     // due beyond horizon → last week
      demByWeek[wi] += d.qty;
    });
    var cumDem = 0, cumSup = 0;
    for (var wi = 0; wi < nW; wi++) {
      cumDem += demByWeek[wi];
      if (CWI >= 0 && wi < CWI) cumSup += (DATA.rows[ri].actualQtys || [])[wi] || 0;
      else cumSup += effQty(ri, wi);
      if (cumDem > cumSup) flags[wi] = true;
    }
    return flags;
  }
  function applyReconcile() {
    document.querySelectorAll('#body td.qty.behind').forEach(function (td) { td.classList.remove('behind'); });
    if (!showReconcile) return;
    DATA.rows.forEach(function (r, ri) {
      var flags = behindWeeks(ri);
      Object.keys(flags).forEach(function (wi) {
        var td = cellAt(ri, +wi);
        if (td) td.classList.add('behind');
      });
    });
  }

  // ── Sorting ───────────────────────────────────────────────────────────
  function onSort(key) {
    if (sortState.key === key) sortState.dir = (sortState.dir === 'asc') ? 'desc' : 'asc';
    else {
      sortState.key = key;
      // Numeric measures default to descending (high first); text defaults ascending.
      var numeric = (key.indexOf('wk:') === 0) ||
        ['unitPrice', 'total', 'totalVal', 'pastQty', 'pastVal'].indexOf(key) >= 0;
      sortState.dir = numeric ? 'desc' : 'asc';
    }
    applySort(); renderSortArrows();
  }
  function sortVal(ri, key) {
    var r = DATA.rows[ri];
    if (key === 'so') return r.so;
    if (key === 'partNo') return r.partNo;
    if (key === 'description') return r.description;
    if (key === 'unitPrice') return r.unitPrice;
    if (key === 'total') return rowTotal(ri).qty;
    if (key === 'totalVal') return rowTotal(ri).val;
    if (key === 'pastQty') return rowMiss(ri);
    if (key === 'pastVal') return rowMiss(ri) * (r.unitPrice || 0);
    if (key.indexOf('wk:') === 0) {
      var wi = +key.slice(3);
      if (sortMeasure === 'val') {
        if (CWI >= 0 && wi < CWI) return (r.actualVals || [])[wi] || 0;
        return effQty(ri, wi) * (r.unitPrice || 0);
      }
      var q = r.qtys[wi]; return q === '' ? 0 : Number(q);
    }
    return '';
  }
  function applySort() {
    if (!sortState.key) return;
    var body = document.getElementById('body');
    var trs = Array.prototype.slice.call(body.querySelectorAll('tr'));
    trs.sort(function (a, b) {
      var va = sortVal(+a.dataset.ri, sortState.key), vb = sortVal(+b.dataset.ri, sortState.key);
      var r = String(va).localeCompare(String(vb), undefined, { numeric:true, sensitivity:'base' });
      return sortState.dir === 'asc' ? r : -r;
    });
    trs.forEach(function (tr) { body.appendChild(tr); });
  }
  function renderSortArrows() {
    document.querySelectorAll('#headRow .sort-ar').forEach(function (s) { s.textContent = ''; });
    if (!sortState.key) return;
    var arrow = sortState.dir === 'asc' ? '▲' : '▼';
    var rollups = ['total', 'totalVal', 'pastQty', 'pastVal'];
    var sel = (rollups.indexOf(sortState.key) >= 0)
      ? '#headRow [data-sort="total"] .sort-ar'
      : '#headRow [data-sort="' + sortState.key + '"] .sort-ar';
    var th = document.querySelector(sel);
    if (th) th.textContent = arrow;
  }

  // ── Inline edit ───────────────────────────────────────────────────────
  function cellAt(ri, wi) { return document.querySelector('#body td.qty[data-ri="' + ri + '"][data-wi="' + wi + '"]'); }

  function startEdit(td) {
    if (!td || td.classList.contains('editing')) return;
    var ri = +td.dataset.ri, wi = +td.dataset.wi, k = ri + ':' + wi;
    if (wi < CWI) return;  // past weeks locked
    var shown = pending.has(k) ? pending.get(k).newQty : (DATA.rows[ri].qtys[wi] === '' ? '' : DATA.rows[ri].qtys[wi]);
    td.classList.add('editing');
    var input = document.createElement('input');
    input.type = 'text'; input.value = (shown === '' ? '' : shown);
    td.innerHTML = ''; td.appendChild(input); input.focus(); input.select();

    var done = false;
    function finish(commit) {
      if (done) return; done = true;
      td.classList.remove('editing');
      var raw = input.value.trim();
      if (!commit) { renderCell(td, ri, wi); return; }
      if (raw !== '' && isNaN(Number(raw))) { renderCell(td, ri, wi); toast('Quantity must be a number.', false); return; }
      var val = raw === '' ? 0 : Number(raw);
      var baseline = DATA.rows[ri].qtys[wi];
      var baseNum = baseline === '' ? 0 : Number(baseline);
      var r = DATA.rows[ri];
      if (val === baseNum) { pending.delete(k); }
      else {
        pending.set(k, { ri:ri, wi:wi, rowNum:r.rowNum, col:DATA.weekCols[wi], weekDate:DATA.weeks[wi],
          oldQty:baseNum, newQty:val, unitPrice:r.unitPrice, partNo:r.partNo, so:r.so });
      }
      renderCell(td, ri, wi);
      var totTd = document.querySelector('#body td.c-tot[data-ri="' + ri + '"]');
      if (totTd) renderTotal(totTd, ri);
      updateHeaderTotals();
      updateMonthBand();
      if (showReconcile) applyReconcile();
      updateCommitUI();
    }
    function go(dr, dc) {
      finish(true);
      var nri = ri + dr, nwi = wi + dc;
      if (nri < 0 || nri >= DATA.rows.length) return;
      if (nwi < CWI || nwi >= DATA.weeks.length) return;
      var ntd = cellAt(nri, nwi);
      if (ntd) startEdit(ntd);
    }
    input.addEventListener('blur', function () { finish(true); });
    input.addEventListener('keydown', function (e) {
      switch (e.key) {
        case 'Enter':      e.preventDefault(); go(1, 0); break;
        case 'Tab':        e.preventDefault(); go(0, e.shiftKey ? -1 : 1); break;
        case 'ArrowDown':  e.preventDefault(); go(1, 0); break;
        case 'ArrowUp':    e.preventDefault(); go(-1, 0); break;
        case 'ArrowRight': e.preventDefault(); go(0, 1); break;
        case 'ArrowLeft':  e.preventDefault(); go(0, -1); break;
        case 'Escape':     e.preventDefault(); finish(false); break;
      }
    });
  }

  // ── Group pending edits ───────────────────────────────────────────────
  function dt(e) { return new Date(e.weekDate).getTime(); }
  function wavg(arr, isLoss) {
    var num = 0, den = 0;
    arr.forEach(function (e) { var w = isLoss ? (e.oldQty - e.newQty) : (e.newQty - e.oldQty); num += dt(e) * w; den += w; });
    return den ? num / den : 0;
  }
  function moveGroup(sources, dests, ownedLosses) {
    var qty = dests.reduce(function (a, e) { return a + (e.newQty - e.oldQty); }, 0);
    var dir = wavg(dests, false) > wavg(sources, true) ? 'PUSH-OUT'
            : (wavg(dests, false) < wavg(sources, true) ? 'PULL-IN' : 'MOVE');
    return { kind:'move', sources:sources, dests:dests, qty:qty, dir:dir,
             partNo:dests[0].partNo, so:dests[0].so, commitCells:ownedLosses.concat(dests) };
  }
  function plainSingle(e) {
    var d = e.newQty - e.oldQty;
    return { kind:'single', synthetic:false, edit:e,
             type:e.newQty === 0 ? 'ZERO' : (d > 0 ? 'INCREASE' : 'DECREASE'),
             partNo:e.partNo, so:e.so, commitCells:[e] };
  }
  function buildGroups() {
    var groups = [];
    var byRow = {};
    pending.forEach(function (e) { (byRow[e.ri] = byRow[e.ri] || []).push(e); });
    Object.keys(byRow).forEach(function (ri) {
      var list = byRow[ri];
      var gains = [], losses = [];
      list.forEach(function (e) { var d = e.newQty - e.oldQty; if (d > 0) gains.push(e); else if (d < 0) losses.push(e); });
      var usedG = {}, usedL = {};
      losses.forEach(function (loss, li) {
        if (usedL[li]) return;
        var mag = loss.oldQty - loss.newQty;
        for (var j = 0; j < gains.length; j++) {
          if (!usedG[j] && (gains[j].newQty - gains[j].oldQty) === mag) {
            usedG[j] = true; usedL[li] = true; groups.push(moveGroup([loss], [gains[j]], [loss])); break;
          }
        }
      });
      var remL = losses.filter(function (_, i) { return !usedL[i]; });
      var remG = gains.filter(function (_, j) { return !usedG[j]; });
      var sumL = remL.reduce(function (a, e) { return a + (e.oldQty - e.newQty); }, 0);
      var sumG = remG.reduce(function (a, e) { return a + (e.newQty - e.oldQty); }, 0);
      if (remL.length && remG.length && sumL >= sumG) {
        var ref = wavg(remL, true);
        var push = remG.filter(function (e) { return dt(e) >= ref; });
        var pull = remG.filter(function (e) { return dt(e) < ref; });
        var remainder = sumL - sumG;
        var lossAssigned = false;
        if (push.length) { var ownP = (remainder === 0 && !lossAssigned) ? remL : []; if (ownP.length) lossAssigned = true; groups.push(moveGroup(remL, push, ownP)); }
        if (pull.length) { var ownPl = (remainder === 0 && !lossAssigned) ? remL : []; if (ownPl.length) lossAssigned = true; groups.push(moveGroup(remL, pull, ownPl)); }
        if (remainder > 0) groups.push({ kind:'single', synthetic:true, type:'DECREASE', qty:remainder, sources:remL, partNo:remL[0].partNo, so:remL[0].so, commitCells:remL });
      } else {
        remL.forEach(function (e) { groups.push(plainSingle(e)); });
        // Unpaired future GAINS: if this row has an open past miss, treat the
        // bump as a recovery PUSH-OUT (missed) from the missed week instead of
        // a plain INCREASE. Draw down the oldest miss first.
        var missPool = missWeeks(ri);
        remG.forEach(function (e) {
          var gain = e.newQty - e.oldQty;
          while (gain > 0 && missPool.length) {
            var m = missPool[0];
            var take = Math.min(gain, m.qty);
            // synthesize a source edit representing the missed week
            var srcEdit = { rowNum:e.rowNum, col:null, weekDate:m.weekDate,
                            oldQty:0, newQty:0, unitPrice:e.unitPrice, partNo:e.partNo, so:e.so };
            var destEdit = { rowNum:e.rowNum, col:e.col, weekDate:e.weekDate,
                             oldQty:e.oldQty, newQty:e.oldQty + take, unitPrice:e.unitPrice, partNo:e.partNo, so:e.so };
            groups.push({ kind:'move', dir:'PUSH-OUT', sources:[srcEdit], dests:[destEdit],
                          qty:take, partNo:e.partNo, so:e.so, pastDue:true, note:'recovery plan (past-due)',
                          commitCells:[e] });
            m.qty -= take; gain -= take;
            if (m.qty <= 0) missPool.shift();
          }
          if (gain > 0) {
            // remaining bump beyond the miss is a normal increase
            var inc = { rowNum:e.rowNum, col:e.col, weekDate:e.weekDate,
                        oldQty:e.newQty - gain, newQty:e.newQty, unitPrice:e.unitPrice, partNo:e.partNo, so:e.so };
            groups.push(plainSingle(inc));
          }
        });
      }
    });
    return groups;
  }

  function fmtDate(str) {
    var d = new Date(str);
    if (isNaN(d.getTime())) return str;
    var m = d.getMonth() + 1, dd = d.getDate(), yy = String(d.getFullYear()).slice(2);
    return (m < 10 ? '0' + m : m) + '/' + (dd < 10 ? '0' + dd : dd) + '/' + yy;
  }
  function wkDelta(fromStr, toStr) {
    var wks = Math.round((new Date(toStr) - new Date(fromStr)) / (7 * 24 * 60 * 60 * 1000));
    return (wks >= 0 ? '+' : '') + wks + ' wk';
  }

  // ── Review modal ──────────────────────────────────────────────────────
  function openReview() {
    if (!pending.size) return;
    commitGroups = buildGroups();
    var list = document.getElementById('reviewList');
    list.innerHTML = '';
    commitGroups.forEach(function (g, i) {
      var info;
      if (g.kind === 'move') {
        var cls = g.dir.toLowerCase().replace('-', '');
        var detail;
        if (g.sources.length === 1 && g.dests.length === 1) {
          var f = g.sources[0].weekDate, t = g.dests[0].weekDate;
          detail = '<b>' + wkDelta(f, t) + '</b>  ·  <b>QTY ' + g.qty + '</b>  FROM ' + fmtDate(f) + ' → TO ' + fmtDate(t);
        } else {
          var srcDate = g.sources[0].weekDate;
          detail = g.dests.map(function (dst) {
            var dstQty = dst.newQty - dst.oldQty;
            return '<b>' + wkDelta(srcDate, dst.weekDate) + '</b>  ·  <b>QTY ' + dstQty + '</b>  FROM ' + fmtDate(srcDate) + ' → TO ' + fmtDate(dst.weekDate);
          }).join('<br>');
        }
        info = '<span class="chip ' + cls + '">' + g.dir + '</span>' +
               '<div class="rv-text"><span class="rv-id">DPN ' + esc(g.partNo) + ' · SO ' + esc(g.so) +
               (g.pastDue ? ' <span class="tag-due">PAST-DUE</span>' : '') + '</span>' +
               '<span class="muted rv-detail">' + detail + '</span></div>';
      } else if (g.synthetic) {
        info = '<span class="chip decrease">DECREASE</span>' +
               '<div class="rv-text">DPN ' + esc(g.partNo) + '  ·  SO ' + esc(g.so) +
               '<br><span class="muted"><b>QTY net −' + g.qty + '</b>  w/e ' +
               g.sources.map(function (e) { return fmtDate(e.weekDate); }).join(', ') + '</span></div>';
      } else {
        var e = g.edit;
        info = '<span class="chip ' + g.type.toLowerCase() + '">' + g.type + '</span>' +
               '<div class="rv-text">DPN ' + esc(g.partNo) + '  ·  SO ' + esc(g.so) +
               '<br><span class="muted"><b>QTY ' + e.oldQty + ' → ' + e.newQty + '</b>  w/e ' + fmtDate(e.weekDate) + '</span></div>';
      }
      var line = document.createElement('div');
      line.className = 'rv-line'; line.id = 'rvline-' + i;
      line.innerHTML = '<div class="rv-info">' + info + '</div><div class="rv-reason">' + reasonControls(i) + '</div>';
      list.appendChild(line);
    });
    commitGroups.forEach(function (g, i) {
      var sel = document.getElementById('rsel-' + i);
      sel.addEventListener('change', function () {
        var txt = document.getElementById('rtxt-' + i);
        if (sel.value === OTHER) { txt.classList.add('show'); txt.focus(); } else { txt.classList.remove('show'); }
      });
    });
    document.getElementById('reviewErr').textContent = '';
    document.getElementById('applyAll').value = '';
    document.getElementById('reviewCount').textContent = commitGroups.length + ' line' + (commitGroups.length === 1 ? '' : 's');
    var b = document.getElementById('saveAllBtn'); b.disabled = false; b.textContent = 'Save all changes';
    document.getElementById('veil').classList.add('open');
  }

  function serEdit(e) {
    return { rowNum:e.rowNum, col:e.col, weekDate:e.weekDate, oldQty:e.oldQty, newQty:e.newQty, unitPrice:e.unitPrice, partNo:e.partNo, so:e.so };

  // ── Save all ──────────────────────────────────────────────────────────
  function saveAll() {
    var resolved = [], missing = false;
    commitGroups.forEach(function (g, i) {
      var reason = resolveReason(i);
      var line = document.getElementById('rvline-' + i);
      if (!reason) { missing = true; line.classList.add('missing'); } else { line.classList.remove('missing'); }
      resolved.push(reason);
    });
    if (missing) { document.getElementById('reviewErr').textContent = 'Add a reason for each highlighted change.'; return; }
    document.getElementById('reviewErr').textContent = '';

    var groups = commitGroups.map(function (g, i) {
      var reason = resolved[i];
      var cells = g.commitCells.map(function (e) { return { rowNum:e.rowNum, col:e.col, newQty:e.newQty }; });
      if (g.kind === 'move') return { kind:'move', rationale:reason, cells:cells, note:g.note || '', sources:g.sources.map(serEdit), dests:g.dests.map(serEdit) };
      if (g.synthetic) return { kind:'synthetic', rationale:reason, cells:cells, qty:g.qty, sources:g.sources.map(serEdit) };
      return { kind:'single', rationale:reason, cells:cells, edit:serEdit(g.edit) };
    });

    var btn = document.getElementById('saveAllBtn');
    btn.disabled = true; btn.textContent = 'Saving…';
    document.getElementById('saveState').textContent = 'Saving…';

    google.script.run
      .withSuccessHandler(function (res) {
        document.getElementById('saveState').textContent = 'Ready';
        if (res && res.ok) {
          pending.forEach(function (e, k) { savedKeys.add(k); DATA.rows[e.ri].qtys[e.wi] = (e.newQty === 0 ? '' : e.newQty); });
          pending.clear(); reRenderAll(); updateHeaderTotals(); updateMonthBand(); updateCommitUI(); closeReview();
          toast(res.applied + ' change' + (res.applied === 1 ? '' : 's') + ' saved and logged.', true);
        } else {
          btn.disabled = false; btn.textContent = 'Save all changes';
          var msg = (res && res.errors && res.errors.length) ? ('Save failed: ' + res.errors[0]) : 'Save failed.';
          document.getElementById('reviewErr').textContent = msg;
        }
      })
      .withFailureHandler(function (e) {
        btn.disabled = false; btn.textContent = 'Save all changes';
        document.getElementById('saveState').textContent = 'Ready';
        document.getElementById('reviewErr').textContent = (e && e.message) ? e.message : 'Save failed. Try again.';
      })
      .submitBatch(groups);
  }
